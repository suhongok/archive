```c++
#include <Arduino.h>

// ===== 설정 =====
const uint32_t BAUD = 115200;

// “문자 간 공백”이 이 시간(ms) 이상이면 한 프레임 끝으로 간주
// 9600bps에서 1바이트 ≈1.04ms → 여유를 줘서 4~8ms 권장
const uint16_t GAP_MS = 2;

const char *HEARTBEAT = "ESP32->RS485 OK\r\n";

void setup() {
  Serial.begin(115200);
  Serial2.begin(BAUD, SERIAL_8N1, 16, 17); // RX=16, TX=17
  Serial.println("\n[Framed RS485 Echo @9600 8N1]");
}

void loop() {
  static uint8_t rxbuf[512];
  static size_t  rxcnt = 0;
  static uint32_t last_rx_ms = 0;

  // ---------- RS485 수신 수집 ----------
  while (Serial2.available()) {
    int c = Serial2.read();
    if (c < 0) break;
    if (rxcnt < sizeof(rxbuf)) rxbuf[rxcnt++] = (uint8_t)c;
    last_rx_ms = millis();
  }

  // 프레임 종료 판단: GAP_MS 이상 추가 수신 없음
  if (rxcnt > 0 && (millis() - last_rx_ms) >= GAP_MS) {
    // (선택) 모니터로 보기
    Serial.write(rxbuf, rxcnt);

    // 자신이 방금 보낸 하트비트가 되돌아온 경우 재에코 방지(무한루프 차단)
    bool is_heartbeat = false;
    if (rxcnt == strlen(HEARTBEAT)) {
      is_heartbeat = (memcmp(rxbuf, HEARTBEAT, rxcnt) == 0);
    }

    // 한 번에 에코 송신
    if (!is_heartbeat) {
      Serial2.write(rxbuf, rxcnt);
      // AutoDirection이 알아서 전환하지만, 상대가 수신할 여유를 조금 줌
      // (특히 USB-RS485 동글/긴 케이블에서 안전)
      delay(1);
    }

    rxcnt = 0; // 버퍼 비우기
  }

  // ---------- USB(Serial) -> RS485 브리지 ----------
  // PC 시리얼 모니터에서 보낸 덩어리를 한 번에 전달
  if (Serial.available()) {
    static uint8_t usbBuf[512];
    size_t n = Serial.readBytes(usbBuf, sizeof(usbBuf));
    if (n > 0) {
      Serial2.write(usbBuf, n);
      // RS485에도 나간 걸 모니터에도 그대로 보여주고 싶으면 아래 줄 주석 해제
      // Serial.write(usbBuf, n);
    }
  }

  // ---------- 1초마다 상태 메시지 (상대 수신 확인) ----------
  static uint32_t t0 = 0;
  if (millis() - t0 > 1000) {
    t0 = millis();
    Serial2.print(HEARTBEAT);
  }
}
```

```c++
#!/usr/bin/env python3
"""
메카넘 휠 하드웨어 테스트 스크립트
mecanum_rs485_hardware.cpp 기반으로 작성됨

사용법: python3 test_wheel.py /dev/ttyACM0

# 기본 테스트 (2초간 각 동작)
python3 test_wheel.py /dev/ttyACM0

# 슬레이브 ID 지정
python3 test_wheel.py /dev/ttyACM0 -i 2

# 테스트 지속시간 조절
python3 test_wheel.py /dev/ttyACM0 -d 3.0

# 연속 모니터링 추가 (10초간)
python3 test_wheel.py /dev/ttyACM0 -m 10

# 도움말
python3 test_wheel.py --help


"""

import minimalmodbus
import sys
import time
import argparse
import threading

class MecanumHardwareTester:
    def __init__(self, port_path, slave_id=1):
        self.port_path = port_path
        self.slave_id = slave_id
        self.motor = None
        self.connected = False
        
        # 하드웨어 스펙 (C++ 코드와 동일)
        self.gear_ratio = 50.0
        self.wheel_diameter_m = 0.2
        self.wheel_circumference = 3.14159 * self.wheel_diameter_m
        
        # 가속도 파라미터 (C++ 코드와 동일)
        self.accel_params = {
            'forward_base_ms': 200,
            'forward_variable_ms': 200,
            'forward_min_ms': 250,
            'forward_max_ms': 600,
            'reverse_base_ms': 150,
            'reverse_variable_ms': 150,
            'reverse_min_ms': 100,
            'reverse_max_ms': 300
        }
        
        print(f"🔧 하드웨어 스펙: 감속기={self.gear_ratio}:1, 바퀴직경={self.wheel_diameter_m*100}cm")
        print(f"🔄 가속도 설정: 전진={self.accel_params['forward_base_ms']}ms, 후진={self.accel_params['reverse_base_ms']}ms")

    def connect(self):
        """모터 연결"""
        try:
            self.motor = minimalmodbus.Instrument(self.port_path, self.slave_id)
            
            # 시리얼 설정 (C++ 코드와 동일)
            self.motor.serial.baudrate = 115200
            self.motor.serial.bytesize = 8
            self.motor.serial.parity = minimalmodbus.serial.PARITY_NONE
            self.motor.serial.stopbits = 1
            self.motor.serial.timeout = 1
            # 추가 안정화 설정
            self.motor.mode = minimalmodbus.MODE_RTU
            # 포트를 계속 유지하여 충돌 가능성 감소
            self.motor.close_port_after_each_call = False
            # 트랜잭션 전 버퍼 정리 (노이즈/에코 제거)
            self.motor.clear_buffers_before_each_transaction = True
            # RS485 어댑터의 로컬 에코 처리 비활성화 (일부 어댑터에서 오류 유발)
            self.motor.handle_local_echo = False
            # pySerial 고유 잠금으로 동시 접근 방지 (지원되는 플랫폼에서만 동작)
            try:
                self.motor.serial.exclusive = True
            except Exception:
                pass
            # 타이밍 여유
            self.motor.serial.inter_byte_timeout = 0.05
            self.motor.serial.write_timeout = 0.5
            try:
                # 잔여 데이터 플러시
                self.motor.serial.reset_input_buffer()
                self.motor.serial.reset_output_buffer()
            except Exception:
                pass
            time.sleep(0.05)
            
            # 연결 테스트
            self.motor.read_register(0x0094, functioncode=3)
            
            self.connected = True
            print(f"✅ {self.port_path} 포트 연결 성공 (슬레이브 ID: {self.slave_id})")
            return True
            
        except Exception as e:
            print(f"❌ 연결 실패: {e}")
            self.connected = False
            return False

    def calculate_accel_time(self, speed, is_forward):
        """가속 시간 계산 (C++ 코드와 동일한 로직)"""
        if is_forward:
            accel_time = int(self.accel_params['forward_base_ms'] + 
                           (speed - 1000.0) / 1000.0 * self.accel_params['forward_variable_ms'])
            return max(self.accel_params['forward_min_ms'], 
                      min(accel_time, self.accel_params['forward_max_ms']))
        else:
            accel_time = int(self.accel_params['reverse_base_ms'] + 
                           (speed - 1000.0) / 1000.0 * self.accel_params['reverse_variable_ms'])
            return max(self.accel_params['reverse_min_ms'], 
                      min(accel_time, self.accel_params['reverse_max_ms']))

    def convert_wheel_velocity_to_motor_rpm(self, wheel_velocity_rad_per_sec):
        """휠 각속도를 모터 RPM으로 변환 (C++ 코드와 동일)"""
        wheel_rpm = wheel_velocity_rad_per_sec * 60.0 / (2.0 * 3.14159)
        motor_rpm = wheel_rpm * self.gear_ratio
        return motor_rpm

    def convert_motor_rpm_to_wheel_velocity(self, motor_rpm):
        """모터 RPM을 휠 각속도로 변환 (C++ 코드와 동일)"""
        wheel_rpm = motor_rpm / self.gear_ratio
        wheel_velocity_rad_per_sec = wheel_rpm * (2.0 * 3.14159) / 60.0
        return wheel_velocity_rad_per_sec

    def write_motor_speed_batch(self, speed):
        """배치 레지스터 쓰기 (C++ 코드와 동일한 방식)"""
        if not self.connected:
            return False
            
        try:
            is_forward = speed >= 0
            abs_speed = abs(speed)
            
            # 가속/감속 시간 계산
            accel_time = self.calculate_accel_time(abs_speed, is_forward)
            
            # 목표 속도 계산 (C++ 코드와 동일한 로직)
            if is_forward:
                speed_val = int(abs_speed)
            else:
                # 후진은 2의 보수 사용
                speed_val = (1 << 32) - int(abs_speed)
            
            high_word = (speed_val >> 16) & 0xFFFF
            low_word = speed_val & 0xFFFF
            
            # 모드 레지스터 설정 (0x0094: 3)
            self.motor.write_register(0x0094, 3, functioncode=6)
            
            # 배치 레지스터 쓰기 (0x009E ~ 0x00A3: 6개 레지스터)
            # C++ 코드와 동일한 구조
            batch_regs = [
                0,           # 0x009E: accel1
                accel_time,  # 0x009F: accel2
                0,           # 0x00A0: accel3
                accel_time,  # 0x00A1: accel4
                high_word,   # 0x00A2: 속도 상위
                low_word     # 0x00A3: 속도 하위
            ]
            
            self.motor.write_registers(0x009E, batch_regs)
            
            print(f"⚙️ 가속시간: {accel_time}ms, 속도: {abs_speed}rpm ({'전진' if is_forward else '후진'})")
            return True
            
        except Exception as e:
            print(f"❌ 속도 설정 실패: {e}")
            return False

    def stop_motor(self):
        """모터 정지 (C++ 코드와 동일)"""
        if not self.connected:
            return False
            
        try:
            # 속도를 0으로 설정
            stop_regs = [0, 0]
            self.motor.write_registers(0x00A2, stop_regs)
            print("🛑 모터 정지")
            return True
            
        except Exception as e:
            print(f"❌ 모터 정지 실패: {e}")
            return False

    def execute_motor_command(self, speed):
        """모터 명령 실행 (C++ 코드와 동일한 로직)"""
        if not self.connected:
            return False
            
        try:
            # 속도가 0이면 정지
            if abs(speed) < 1e-6:
                if not self.stop_motor():
                    return False
                return True
            
            # 속도 설정
            if not self.write_motor_speed_batch(speed):
                return False
            
            # 모터 실행 명령 (0x0091 레지스터)
            run_cmd = 8  # 실행: 8
            self.motor.write_register(0x0091, run_cmd, functioncode=6)
            
            direction = "전진" if speed > 0 else "후진"
            print(f"⚡ 모터 {direction}: {abs(speed)}rpm")
            return True
            
        except Exception as e:
            print(f"❌ 모터 명령 실행 실패: {e}")
            return False

    def read_motor_status(self):
        """모터 상태 읽기"""
        if not self.connected:
            return None
            
        try:
            # 모드 레지스터 읽기
            mode = self.motor.read_register(0x0094, functioncode=3)
            
            # 속도 레지스터 읽기
            speed_high = self.motor.read_register(0x00A2, functioncode=3)
            speed_low = self.motor.read_register(0x00A3, functioncode=3)
            
            # 32비트 속도 값으로 결합
            velocity_raw = (speed_high << 16) | speed_low
            
            # 2의 보수 처리 (후진 속도)
            if velocity_raw & (1 << 31):
                velocity_raw = velocity_raw - (1 << 32)
            
            return {
                'mode': mode,
                'speed_raw': velocity_raw,
                'speed_rpm': abs(velocity_raw),
                'direction': '전진' if velocity_raw >= 0 else '후진'
            }
            
        except Exception as e:
            print(f"❌ 상태 읽기 실패: {e}")
            return None

    def test_motor_sequence(self, duration=2.0):
        """모터 테스트 시퀀스 실행"""
        if not self.connected:
            print("❌ 모터가 연결되지 않았습니다")
            return False
        
        print(f"\n🚀 모터 테스트 시퀀스 시작 (지속시간: {duration}초)")
        print("-" * 50)
        
        try:
            # 1. 전진 테스트
            print("1️⃣ 전진 테스트...")
            if not self.execute_motor_command(500):  # 500rpm
                print("💥 전진 명령 실패로 시퀀스를 중단합니다")
                return False
            time.sleep(duration)
            
            # 2. 정지
            print("2️⃣ 정지...")
            if not self.stop_motor():
                print("💥 정지 실패로 시퀀스를 중단합니다")
                return False
            time.sleep(0.5)
            
            # 3. 후진 테스트
            print("3️⃣ 후진 테스트...")
            if not self.execute_motor_command(-500):  # -500rpm
                print("💥 후진 명령 실패로 시퀀스를 중단합니다")
                return False
            time.sleep(duration)
            
            # 4. 정지
            print("4️⃣ 정지...")
            if not self.stop_motor():
                print("💥 정지 실패로 시퀀스를 중단합니다")
                return False
            time.sleep(0.5)
            
            # 5. 고속 전진 테스트
            print("5️⃣ 고속 전진 테스트...")
            if not self.execute_motor_command(1000):  # 1000rpm
                print("💥 고속 전진 명령 실패로 시퀀스를 중단합니다")
                return False
            time.sleep(duration)
            
            # 6. 최종 정지
            print("6️⃣ 최종 정지...")
            if not self.stop_motor():
                print("💥 최종 정지 실패")
                return False
            
            print("✅ 테스트 시퀀스 완료!")
            return True
            
        except Exception as e:
            print(f"❌ 테스트 시퀀스 실패: {e}")
            return False

    def continuous_monitoring(self, duration=10):
        """연속 모니터링"""
        if not self.connected:
            print("❌ 모터가 연결되지 않았습니다")
            return
        
        print(f"\n📊 연속 모니터링 시작 (지속시간: {duration}초)")
        print("-" * 50)
        
        start_time = time.time()
        try:
            while time.time() - start_time < duration:
                status = self.read_motor_status()
                if status:
                    print(f"⏰ {time.strftime('%H:%M:%S')} - "
                          f"모드: {status['mode']}, "
                          f"속도: {status['speed_rpm']}rpm ({status['direction']})")
                time.sleep(0.5)
        except KeyboardInterrupt:
            print("\n⏹️ 모니터링 중단됨")
        
        print("✅ 모니터링 완료")

def main():
    parser = argparse.ArgumentParser(description='메카넘 휠 하드웨어 테스트')
    parser.add_argument('port', help='USB 포트 경로 (예: /dev/ttyACM0)')
    parser.add_argument('-i', '--slave-id', type=int, default=1, 
                       help='슬레이브 ID (기본값: 1)')
    parser.add_argument('-d', '--duration', type=float, default=2.0, 
                       help='각 테스트 지속시간 (초, 기본값: 2.0)')
    parser.add_argument('-m', '--monitor', type=int, default=0,
                       help='연속 모니터링 시간 (초, 0=비활성, 기본값: 0)')
    
    args = parser.parse_args()
    
    if not args.port.startswith('/dev/'):
        print("❌ 올바른 포트 경로를 입력하세요 (예: /dev/ttyACM0)")
        sys.exit(1)
    
    print(f"🚀 메카넘 휠 하드웨어 테스트 시작")
    print(f"📍 포트: {args.port}")
    print(f"🆔 슬레이브 ID: {args.slave_id}")
    print(f"⏱️ 테스트 지속시간: {args.duration}초")
    if args.monitor > 0:
        print(f"📊 모니터링 시간: {args.monitor}초")
    print("-" * 50)
    
    # 테스터 생성 및 연결
    tester = MecanumHardwareTester(args.port, args.slave_id)
    
    if not tester.connect():
        print("💥 연결 실패로 테스트를 중단합니다")
        sys.exit(1)
    
    try:
        # 기본 테스트 시퀀스 실행
        if not tester.test_motor_sequence(args.duration):
            print("💥 테스트 시퀀스 실패")
            sys.exit(1)
        
        # 연속 모니터링 (옵션)
        if args.monitor > 0:
            tester.continuous_monitoring(args.monitor)
        
        print("🎉 모든 테스트가 성공적으로 완료되었습니다!")
        
    except KeyboardInterrupt:
        print("\n⏹️ 사용자에 의해 테스트가 중단되었습니다")
        tester.stop_motor()
    except Exception as e:
        print(f"💥 예상치 못한 오류: {e}")
        tester.stop_motor()
        sys.exit(1)
    finally:
        # 안전 정지
        tester.stop_motor()

if __name__ == "__main__":
    main()
```
