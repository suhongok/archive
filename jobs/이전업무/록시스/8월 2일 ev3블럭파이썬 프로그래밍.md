#!/usr/bin/env pybricks-micropython
from pybricks.hubs import EV3Brick
from pybricks.ev3devices import (Motor, TouchSensor, ColorSensor,
                                 InfraredSensor, UltrasonicSensor, GyroSensor)
from pybricks.parameters import Port, Stop, Direction, Button, Color
from pybricks.tools import wait, StopWatch, DataLog
from pybricks.robotics import DriveBase
from pybricks.media.ev3dev import SoundFile, ImageFile


# EV3 브릭 인스턴스 생성
ev3 = EV3Brick()

# 포트 A, B, C에 연결된 모터 인스턴스 생성
motor_a = Motor(Port.A)
motor_b = Motor(Port.B)
motor_c = Motor(Port.C)

# 모터를 앞으로 회전시키는 함수
def run_motors():
    # 모터를 앞으로 회전
    motor_a.run(500)  # 속도 500의 앞으로 회전
    motor_b.run(500)
    motor_c.run(500)
    wait(1000)  # 2초 동안 대기
    # 모터 정지
    motor_a.stop()
    motor_b.stop()
    motor_c.stop()
    
    motor_a.run(-500)  # 속도 500의 앞으로 회전
    motor_b.run(-500)
    motor_c.run(-500)
    wait(1000)  # 2초 동안 대기
    
    motor_a.stop()
    motor_b.stop()
    motor_c.stop()
    

# 모터 실행 함수 호출
run_motors()

#!/usr/bin/env pybricks-micropython
from pybricks.hubs import EV3Brick
from pybricks.ev3devices import (Motor, TouchSensor, ColorSensor,
                                 InfraredSensor, UltrasonicSensor, GyroSensor)
from pybricks.parameters import Port, Stop, Direction, Button, Color
from pybricks.tools import wait, StopWatch, DataLog
from pybricks.robotics import DriveBase
from pybricks.media.ev3dev import SoundFile, ImageFile

# EV3 브릭 인스턴스 생성
ev3 = EV3Brick()

# 포트 A에 연결된 모터 인스턴스 생성
motor_a = Motor(Port.A)
motor_b = Motor(Port.B)
motor_c = Motor(Port.C)

#초음파 센서 정의
ir_sensor = UltrasonicSensor(Port.S2)

# 버튼을 누를 때까지 기다린 후 모터를 앞으로 회전시키는 함수
def run_motor_on_button_press():
    while True:
        if Button.UP in ev3.buttons.pressed():
            wait(10)  # 버튼 상태를 확인하기 위해 매우 짧은 시간 동안 대기
            # 버튼이 눌릴 때까지 기다린 후 모터를 앞으로 회전
            motor_a.run(500)  # 속도 500의 앞으로 회전
            motor_b.run(500)  # 속도 500의 앞으로 회전
            wait(1000)  # 2초 동안 대기
            motor_a.stop()  # 모터 정지
            motor_b.stop()  # 모터 정지
        if Button.DOWN in ev3.buttons.pressed():
            wait(10)  # 버튼 상태를 확인하기 위해 매우 짧은 시간 동안 대기
            # 버튼이 눌릴 때까지 기다린 후 모터를 앞으로 회전
            motor_a.run(-500)  # 속도 500의 앞으로 회전
            motor_b.run(-500)  # 속도 500의 앞으로 회전
            wait(1000)  # 2초 동안 대기
            motor_a.stop()  # 모터 정지
            motor_b.stop()  # 모터 정지
        if Button.LEFT in ev3.buttons.pressed():
            wait(10)  # 버튼 상태를 확인하기 위해 매우 짧은 시간 동안 대기
            # 버튼이 눌릴 때까지 기다린 후 모터를 앞으로 회전
            motor_c.run(200)  # 속도 500의 앞으로 회전
            wait(1000)  # 2초 동안 대기
            motor_c.stop()  # 모터 정지
        if Button.RIGHT in ev3.buttons.pressed():
            wait(10)  # 버튼 상태를 확인하기 위해 매우 짧은 시간 동안 대기
            # 버튼이 눌릴 때까지 기다린 후 모터를 앞으로 회전
            motor_c.run(-200)  # 속도 500의 앞으로 회전
            wait(1000)  # 2초 동안 대기
            motor_c.stop()  # 모터 정지
        if Button.CENTER in ev3.buttons.pressed():
            wait(10)  # 버튼 상태를 확인하기 위해 매우 짧은 시간 동안 대기
            # 버튼이 눌릴 때까지 기다린 후 모터를 앞으로 회전
            motor_a.stop()# 회전
            motor_b.stop()# 회전
            motor_c.stop()  # 모터 정지
        distance = ir_sensor.distance()
        if distance < 3:
            ev3.speaker.beep()
            wait(1000)
# 모터 실행 함수 호출
run_motor_on_button_press()
